package com.truesign.blockchain;

import com.truesign.dto.request.RegisterPayeeRequest;
import com.truesign.entity.PayeeRecord;
import com.truesign.repository.PayeeRepository;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.web3j.abi.FunctionEncoder;
import org.web3j.abi.datatypes.Address;
import org.web3j.abi.datatypes.Function;
import org.web3j.protocol.Web3j;
import org.web3j.protocol.core.DefaultBlockParameterName;
import org.web3j.protocol.core.methods.response.Transaction;
import org.web3j.utils.Numeric;

import java.math.BigInteger;
import java.time.Instant;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

@Service
public class PayeeRegistryService {

    private final Web3j web3j;
    private final PayeeRepository repository;

    @Value("${truesign.contracts.payee-registry-address}")
    private String registryAddress;

    @Value("${truesign.blockchain.chain-id}")
    private long expectedChainId;

    public PayeeRegistryService(Web3j web3j, PayeeRepository repository) {
        this.web3j = web3j;
        this.repository = repository;
    }

    public Map<String, Object> confirmRegistration(RegisterPayeeRequest request)
            throws Exception {

        String account = normalize(request.account());
        String payee = normalize(request.payee());
        String txHash = request.transactionHash();

        if (account.equals(payee)) {
            throw new IllegalArgumentException("An account cannot register itself as a payee.");
        }

        long chainId = web3j.ethChainId().send().getChainId().longValueExact();

        if (chainId != expectedChainId) {
            throw new IllegalStateException(
                    "Wrong blockchain network. Expected chain " + expectedChainId +
                    " but connected to " + chainId
            );
        }

        var receiptResponse = web3j.ethGetTransactionReceipt(txHash).send();

        var receipt = receiptResponse.getTransactionReceipt()
                .orElseThrow(() -> new IllegalArgumentException(
                        "Transaction is not mined or does not exist."
                ));

        if (!"0x1".equalsIgnoreCase(receipt.getStatus())) {
            throw new IllegalArgumentException("Payee registration transaction failed.");
        }

        Transaction transaction = web3j.ethGetTransactionByHash(txHash)
                .send()
                .getTransaction()
                .orElseThrow(() -> new IllegalArgumentException(
                        "Transaction details could not be retrieved."
                ));

        if (!account.equalsIgnoreCase(transaction.getFrom())) {
            throw new IllegalArgumentException(
                    "Transaction sender does not match the requested account."
            );
        }

        if (transaction.getTo() == null ||
                !registryAddress.equalsIgnoreCase(transaction.getTo())) {
            throw new IllegalArgumentException(
                    "Transaction was not sent to the configured PayeeRegistry."
            );
        }

        Function registerFunction = new Function(
                "registerPayee",
                List.of(new Address(payee)),
                List.of()
        );

        String expectedInput = FunctionEncoder.encode(registerFunction);

        if (!expectedInput.equalsIgnoreCase(transaction.getInput())) {
            throw new IllegalArgumentException(
                    "Transaction calldata does not match registerPayee(payee)."
            );
        }

        BigInteger activationTimestamp = readUint256(
                "activationTimestamp",
                List.of(new Address(account), new Address(payee))
        );

        if (activationTimestamp.signum() <= 0) {
            throw new IllegalStateException(
                    "Registration transaction succeeded but the contract has no activation timestamp."
            );
        }

        PayeeRecord record = repository
                .findByAccountAndPayee(account, payee)
                .orElseGet(PayeeRecord::new);

        record.setAccount(account);
        record.setPayee(payee);
        record.setTransactionHash(txHash);
        record.setActivationTimestamp(activationTimestamp.toString());
        record.setCreatedAt(Instant.now());

        repository.save(record);

        return getPayeeStatus(account, payee);
    }

    public Map<String, Object> getPayeeStatus(String account, String payee)
            throws Exception {

        account = normalize(account);
        payee = normalize(payee);

        boolean registered = readBool(
                "isPayeeRegistered",
                List.of(new Address(account), new Address(payee))
        );

        boolean active = readBool(
                "isPayeeActive",
                List.of(new Address(account), new Address(payee))
        );

        BigInteger activationTimestamp = readUint256(
                "activationTimestamp",
                List.of(new Address(account), new Address(payee))
        );

        Map<String, Object> result = new LinkedHashMap<>();
        result.put("account", account);
        result.put("payee", payee);
        result.put("registered", registered);
        result.put("active", active);
        result.put("activationTimestamp", activationTimestamp.toString());
        result.put("activationTime", activationTimestamp.signum() > 0
                ? Instant.ofEpochSecond(activationTimestamp.longValue()).toString()
                : null);
        result.put("registryAddress", registryAddress);
        result.put("chainId", expectedChainId);

        repository.findByAccountAndPayee(account, payee).ifPresent(record -> {
            result.put("registrationTransactionHash", record.getTransactionHash());
            result.put("backendRecordedAt", record.getCreatedAt());
        });

        return result;
    }

    private boolean readBool(String functionName, List<org.web3j.abi.datatypes.Type> inputs)
            throws Exception {
        String output = callView(functionName, inputs);
        return new BigInteger(output, 16).signum() != 0;
    }

    private BigInteger readUint256(
            String functionName,
            List<org.web3j.abi.datatypes.Type> inputs
    ) throws Exception {
        String output = callView(functionName, inputs);
        return new BigInteger(output, 16);
    }

    private String callView(
            String functionName,
            List<org.web3j.abi.datatypes.Type> inputs
    ) throws Exception {

        Function function = new Function(functionName, inputs, List.of());
        String data = FunctionEncoder.encode(function);

        var response = web3j.ethCall(
                org.web3j.protocol.core.methods.request.Transaction.createEthCallTransaction(
                        null, registryAddress, data
                ),
                DefaultBlockParameterName.LATEST
        ).send();

        if (response.hasError()) {
            throw new IllegalStateException(
                    "PayeeRegistry call failed: " + response.getError().getMessage()
            );
        }

        String value = Numeric.cleanHexPrefix(response.getValue());

        if (value == null || value.isBlank()) {
            throw new IllegalStateException("PayeeRegistry returned empty data.");
        }

        return value;
    }

    private String normalize(String address) {
        if (address == null ||
                !address.matches("(?i)^0x[a-f0-9]{40}$")) {
            throw new IllegalArgumentException("Invalid Ethereum address.");
        }
        return address.toLowerCase();
    }
}

package com.truesign.service;

import com.google.gson.Gson;
import com.google.gson.GsonBuilder;
import com.google.gson.JsonObject;
import com.google.gson.JsonParser;
import com.truesign.dto.response.ProofBundleResponse;
import com.truesign.dto.response.VerifierVerdictDto;
import com.truesign.entity.PaymentPassport;
import com.truesign.entity.ProofBundle;
import com.truesign.repository.ProofBundleRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.Instant;
import java.util.List;
import java.util.UUID;

@Service
public class ProofBundleService {

    private final ProofBundleRepository repository;
    private final Gson gson;

    public ProofBundleService(ProofBundleRepository repository) {
        this.repository = repository;
        this.gson = new GsonBuilder().create();
    }

    @Transactional
    public ProofBundleResponse createProofBundle(
            PaymentPassport passport,
            List<VerifierVerdictDto> verdicts,
            String overallVerdict
    ) {
        if (passport == null || verdicts == null ||
                overallVerdict == null || overallVerdict.isBlank()) {
            throw new IllegalArgumentException(
                    "Passport, verifier results and verdict are required."
            );
        }

        Instant now = Instant.now();

        JsonObject proof = new JsonObject();
        proof.addProperty("proofVersion", "2.0");
        proof.addProperty("passportId", passport.getId());
        proof.addProperty("passportHash", passport.getPassportHash());
        proof.addProperty("intentHash", passport.getIntentHash());
        proof.addProperty("applicationTransactionHash",
                passport.getTransactionHash());
        proof.addProperty("account", passport.getAccount());
        proof.addProperty("chainId", passport.getChainId());
        proof.addProperty("overallVerdict", overallVerdict);
        proof.add("verifierResults", gson.toJsonTree(verdicts));
        proof.addProperty("createdAt", now.toString());

        ProofBundle bundle = new ProofBundle();
        bundle.setId(UUID.randomUUID().toString());
        bundle.setPassportId(passport.getId());
        bundle.setPassportHash(passport.getPassportHash());
        bundle.setOverallVerdict(overallVerdict);
        bundle.setProofPayload(gson.toJson(proof));
        bundle.setCreatedAt(now);

        return toResponse(repository.save(bundle));
    }

    @Transactional
    public ProofBundleResponse attachAttestation(
            String passportId,
            String safeTxHash,
            String digest,
            long expiry,
            List<String> signers,
            List<String> signatures,
            String attestationTransactionHash
    ) {
        ProofBundle bundle = latestEntity(passportId);

        JsonObject proof = parse(bundle);

        JsonObject authorization = new JsonObject();
        authorization.addProperty("safeTransactionHash", safeTxHash);
        authorization.addProperty("eip712Digest", digest);
        authorization.addProperty("expiry", expiry);
        authorization.addProperty("attestationTransactionHash",
                attestationTransactionHash);
        authorization.add("signers", gson.toJsonTree(signers));
        authorization.add("signatures", gson.toJsonTree(signatures));
        authorization.addProperty("status", "CONFIRMED_ON_CHAIN");
        authorization.addProperty("recordedAt", Instant.now().toString());

        proof.add("safeAuthorization", authorization);

        bundle.setProofPayload(gson.toJson(proof));

        return toResponse(repository.save(bundle));
    }

    @Transactional
    public ProofBundleResponse attachExecutionReceipt(
            String passportId,
            ExecutionReceiptService.ExecutionReceipt receipt
    ) {
        if (receipt == null) {
            throw new IllegalArgumentException("Execution receipt is required.");
        }

        ProofBundle bundle = latestEntity(passportId);
        JsonObject proof = parse(bundle);

        JsonObject execution = new JsonObject();
        execution.addProperty("transactionHash", receipt.transactionHash());
        execution.addProperty(
                "blockNumber",
                receipt.blockNumber() == null
                        ? null
                        : receipt.blockNumber().toString()
        );
        execution.addProperty("successful", receipt.successful());
        execution.addProperty("receiptStatus", receipt.status());
        execution.addProperty(
                "gasUsed",
                receipt.gasUsed() == null
                        ? null
                        : receipt.gasUsed().toString()
        );
        execution.addProperty(
                "executedAt",
                receipt.executedAt() == null
                        ? null
                        : receipt.executedAt().toString()
        );
        execution.addProperty(
                "outcome",
                receipt.successful() ? "EXECUTED_SUCCESSFULLY" : "EXECUTION_FAILED"
        );
        execution.addProperty("recordedAt", Instant.now().toString());

        proof.add("executionReceipt", execution);

        bundle.setProofPayload(gson.toJson(proof));

        return toResponse(repository.save(bundle));
    }

    @Transactional(readOnly = true)
    public ProofBundleResponse getLatestProofBundle(String passportId) {
        return toResponse(latestEntity(passportId));
    }

    private ProofBundle latestEntity(String passportId) {
        if (passportId == null || passportId.isBlank()) {
            throw new IllegalArgumentException("Passport ID is required.");
        }

        return repository
                .findFirstByPassportIdOrderByCreatedAtDesc(passportId)
                .orElseThrow(() -> new IllegalArgumentException(
                        "Proof bundle not found for passport: " + passportId
                ));
    }

    private JsonObject parse(ProofBundle bundle) {
        try {
            return JsonParser.parseString(bundle.getProofPayload())
                    .getAsJsonObject();
        } catch (Exception exception) {
            throw new IllegalStateException(
                    "Stored proof payload is invalid JSON.", exception
            );
        }
    }

    private ProofBundleResponse toResponse(ProofBundle bundle) {
        return new ProofBundleResponse(
                bundle.getId(),
                bundle.getPassportId(),
                bundle.getPassportHash(),
                bundle.getOverallVerdict(),
                bundle.getProofPayload(),
                bundle.getCreatedAt()
        );
    }
}

package com.truesign.service;

import org.springframework.stereotype.Service;
import org.web3j.protocol.Web3j;
import org.web3j.protocol.core.DefaultBlockParameterNumber;
import org.web3j.protocol.core.methods.response.EthBlock;
import org.web3j.protocol.core.methods.response.TransactionReceipt;

import java.math.BigInteger;
import java.time.Instant;
import java.util.Optional;

@Service
public class ExecutionReceiptService {

    private final Web3j web3j;

    public ExecutionReceiptService(Web3j web3j) {
        this.web3j = web3j;
    }

    /**
     * Retrieves the execution receipt for a blockchain transaction.
     */
    public Optional<ExecutionReceipt> getExecutionReceipt(
            String transactionHash) {

        if (transactionHash == null ||
                !transactionHash.matches("0x[0-9a-fA-F]{64}")) {
            throw new IllegalArgumentException(
                    "Invalid Ethereum transaction hash");
        }

        try {
            var response = web3j
                    .ethGetTransactionReceipt(transactionHash)
                    .send();

            Optional<TransactionReceipt> optionalReceipt =
                    response.getTransactionReceipt();

            if (optionalReceipt.isEmpty()) {
                return Optional.empty();
            }

            TransactionReceipt receipt = optionalReceipt.get();

            BigInteger blockNumber = receipt.getBlockNumber();
            BigInteger gasUsed = receipt.getGasUsed();

            Instant executedAt = null;

            if (blockNumber != null) {
                EthBlock blockResponse = web3j
                        .ethGetBlockByNumber(
                                new DefaultBlockParameterNumber(blockNumber),
                                false)
                        .send();

                EthBlock.Block block = blockResponse.getBlock();

                if (block != null && block.getTimestamp() != null) {
                    executedAt = Instant.ofEpochSecond(
                            block.getTimestamp().longValueExact());
                }
            }

            boolean successful =
                    "0x1".equalsIgnoreCase(receipt.getStatus()) ||
                            "0x01".equalsIgnoreCase(receipt.getStatus());

            return Optional.of(new ExecutionReceipt(
                    receipt.getTransactionHash(),
                    blockNumber,
                    successful,
                    receipt.getStatus(),
                    gasUsed,
                    executedAt
            ));

        } catch (Exception exception) {
            throw new IllegalStateException(
                    "Failed to retrieve blockchain execution receipt: "
                            + exception.getMessage(),
                    exception
            );
        }
    }

    /**
     * Structured execution receipt returned by this service.
     */
    public record ExecutionReceipt(
            String transactionHash,
            BigInteger blockNumber,
            boolean successful,
            String status,
            BigInteger gasUsed,
            Instant executedAt
    ) {}
}

package com.truesign.blockchain;

import com.truesign.dto.response.VerifierVerdictDto;

import com.truesign.entity.VerificationRecord;

import com.google.gson.Gson;
import com.google.gson.GsonBuilder;
import com.google.gson.JsonDeserializer;
import com.google.gson.reflect.TypeToken;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.web3j.abi.FunctionEncoder;
import org.web3j.abi.TypeReference;
import org.web3j.abi.datatypes.DynamicArray;
import org.web3j.abi.datatypes.DynamicBytes;
import org.web3j.abi.datatypes.Function;
import org.web3j.abi.datatypes.Type;
import org.web3j.abi.datatypes.generated.Bytes32;
import org.web3j.abi.datatypes.generated.Uint256;

@Service
public class TrueSignGuardService {

    private final Web3j web3j;
    private final String guardAddress;
    private final String submitterPrivateKey;

    private final PaymentPassportRepository passportRepository;
    private final VerificationRecordRepository verificationRepository;
    private final SafeTransactionService safeTransactionService;
    private final VerifierQuorumService quorumService;
    private final ProofBundleService proofBundleService;

    private final Gson gson;

    public TrueSignGuardService(
            Web3j web3j,
            PaymentPassportRepository passportRepository,
            VerificationRecordRepository verificationRepository,
            SafeTransactionService safeTransactionService,
            VerifierQuorumService quorumService,
            ProofBundleService proofBundleService,
            @Value("${truesign.contracts.guard-address}") String guardAddress,
            @Value("${truesign.blockchain.submitter-private-key:}") String submitterPrivateKey
    ) {
        this.web3j = web3j;
        this.passportRepository = passportRepository;
        this.verificationRepository = verificationRepository;
        this.safeTransactionService = safeTransactionService;
        this.quorumService = quorumService;
        this.proofBundleService = proofBundleService;
        this.guardAddress = guardAddress;
        this.submitterPrivateKey = submitterPrivateKey;

        this.gson = new GsonBuilder()
                .registerTypeAdapter(
                        Instant.class,
                        (JsonDeserializer<Instant>) (json, type, context) ->
                                Instant.parse(json.getAsString())
                )
                .create();
    }

    public record GuardAttestationResult(
            String passportId,
            String safeTransactionHash,
            String digest,
            long expiry,
            List<String> signers,
            String attestationTransactionHash,
            String status
    ) {}

    public GuardAttestationResult authorizePassport(
            String passportId
    ) throws Exception {

        if (submitterPrivateKey == null || submitterPrivateKey.isBlank()) {
            throw new IllegalStateException(
                    "TRUESIGN_SUBMITTER_PRIVATE_KEY is not configured."
            );
        }

        PaymentPassport passport = passportRepository.findById(passportId)
                .orElseThrow(() -> new IllegalArgumentException(
                        "Passport not found: " + passportId
                ));

        if (!"GREEN".equalsIgnoreCase(passport.getStatus())
                && !"CREATED".equalsIgnoreCase(passport.getStatus())) {
            throw new SecurityException(
                    "Passport is not eligible for Guard authorization."
            );
        }

        List<VerificationRecord> records =
                verificationRepository.findByPassportIdOrderByCreatedAtDesc(
                        passportId
                );

        if (records.isEmpty()) {
            throw new IllegalStateException(
                    "Run transaction verification before requesting attestation."
            );
        }

        VerificationRecord latest = records.get(0);

        if (!"GREEN".equalsIgnoreCase(latest.getOverallVerdict())) {
            throw new SecurityException(
                    "Only a GREEN verification result can be authorized."
            );
        }

        java.lang.reflect.Type verdictType =
                new TypeToken<List<VerifierVerdictDto>>() {}.getType();

        List<VerifierVerdictDto> verdicts =
                gson.fromJson(latest.getResultsPayload(), verdictType);

        if (verdicts == null || verdicts.size() != 3) {
            throw new SecurityException(
                    "Three verifier results are required."
            );
        }

        // Calculate the exact Safe transaction hash.
        String safeTxHash =
                safeTransactionService.calculateSafeTransactionHash(passport);

        // 30-minute validity, within Guard's one-hour maximum.
        long expiry = Instant.now().getEpochSecond() + 1800L;

        // Generate and verify the 2-of-3 signatures.
        VerifierQuorumService.QuorumAttestation quorum =
                quorumService.createQuorum(
                        verdicts,
                        latest.getOverallVerdict(),
                        safeTxHash,
                        expiry
                );

        // Submit the two signatures to the deployed Guard.
        String attestationTxHash = submitAttestation(
                safeTxHash,
                expiry,
                quorum.signatures()
        );

        // Persist the authorization proof.
        proofBundleService.attachAttestation(
                passportId,
                safeTxHash,
                quorum.digest(),
                expiry,
                quorum.signers(),
                quorum.signatures(),
                attestationTxHash
        );

        return new GuardAttestationResult(
                passportId,
                safeTxHash,
                quorum.digest(),
                expiry,
                quorum.signers(),
                attestationTxHash,
                "ATTESTATION_SUBMITTED"
        );
    }

    private String submitAttestation(
            String safeTxHash,
            long expiry,
            List<String> signatures
    ) throws Exception {

        if (signatures == null || signatures.size() != 2) {
            throw new SecurityException("Exactly two signatures are required.");
        }

        List<DynamicBytes> signatureBytes = signatures.stream()
                .map(value -> new DynamicBytes(
                        Numeric.hexStringToByteArray(value)
                ))
                .toList();

        DynamicArray<DynamicBytes> signatureArray =
                new DynamicArray<>(DynamicBytes.class, signatureBytes);

        Function function = new Function(
                "submitAttestation",
                List.of(
                        new Bytes32(Numeric.hexStringToByteArray(safeTxHash)),
                        new Uint256(BigInteger.valueOf(expiry)),
                        signatureArray
                ),
                List.of()
        );

        String calldata = FunctionEncoder.encode(function);

        Credentials credentials = Credentials.create(
                Numeric.cleanHexPrefix(submitterPrivateKey.trim())
        );

        BigInteger chainId = web3j.ethChainId()
                .send()
                .getChainId();

        RawTransactionManager manager =
                new RawTransactionManager(
                        web3j,
                        credentials,
                        chainId.longValueExact()
                );

        BigInteger gasPrice = web3j.ethGasPrice()
                .send()
                .getGasPrice();

        EthSendTransaction response = manager.sendTransaction(
                gasPrice,
                BigInteger.valueOf(600_000L),
                guardAddress,
                calldata,
                BigInteger.ZERO
        );

        if (response.hasError()) {
            throw new IllegalStateException(
                    "Guard attestation submission failed: "
                            + response.getError().getMessage()
            );
        }

        String transactionHash = response.getTransactionHash();

        if (transactionHash == null || transactionHash.isBlank()) {
            throw new IllegalStateException(
                    "Guard submission returned no transaction hash."
            );
        }

        TransactionReceipt receipt =
                new PollingTransactionReceiptProcessor(
                        web3j,
                        1000,
                        60
                ).waitForTransactionReceipt(transactionHash);

        if (!"0x1".equalsIgnoreCase(receipt.getStatus())) {
            throw new IllegalStateException(
                    "Guard rejected the attestation transaction. Receipt status: "
                            + receipt.getStatus()
            );
        }

        return transactionHash;
    }
}
