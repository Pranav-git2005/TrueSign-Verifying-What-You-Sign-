package com.truesign.dto.request;

import jakarta.validation.constraints.Min;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Pattern;
import jakarta.validation.constraints.Size;

public record ConstraintsDto(

        @NotBlank(message = "Maximum amount is required")
        @Pattern(regexp = "^[0-9]+(\\.[0-9]+)?$")
        String maxAmount,

        @NotBlank(message = "Allowed target is required")
        @Pattern(regexp = "^0x[a-fA-F0-9]{40}$")
        String allowedTarget,

        @Pattern(regexp = "^0x([a-fA-F0-9]{8})?$")
        String allowedSelector,

        @Pattern(regexp = "^(0x)?([a-fA-F0-9]{64})?$")
        String expectedCodeHash,

        boolean allowDelegateCall,

        boolean allowUnlimitedApproval,

        @NotBlank(message = "Daily limit is required")
        @Pattern(regexp = "^[0-9]+(\\.[0-9]+)?$")
        String dailyLimit,

        @Min(value = 0, message = "Cooldown cannot be negative")
        long cooldownSeconds

) {}

===== CreatePassportRequest.java =====

package com.truesign.dto.request;

import jakarta.validation.Valid;
import jakarta.validation.constraints.Future;
import jakarta.validation.constraints.Min;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Pattern;

import java.time.Instant;

public record CreatePassportRequest(

        @NotBlank(message = "Account address is required")
        @Pattern(regexp = "^0x[a-fA-F0-9]{40}$")
        String account,

        @NotBlank(message = "Mode is required")
        @Pattern(regexp = "^(SAFE_MULTISIG|SINGLE_KEY_EOA)$")
        String mode,

        @Min(value = 1, message = "Chain ID must be positive")
        long chainId,

        @NotNull(message = "Passport expiry is required")
        @Future(message = "Passport expiry must be in the future")
        Instant expiresAt,

        @NotNull
        @Valid
        IntentDto intent,

        @NotNull
        @Valid
        ConstraintsDto constraints,

        @NotNull
        @Valid
        TransactionProposalDto transaction

) {}

===== IntentDto.java =====

package com.truesign.dto.request;

import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Pattern;
import jakarta.validation.constraints.Size;

public record IntentDto(

        @NotBlank(message = "Payment purpose is required")
        @Size(max = 100)
        String purpose,

        @NotBlank(message = "Payment summary is required")
        @Size(max = 500)
        String summary,

        @NotBlank(message = "Intended recipient is required")
        @Pattern(
                regexp = "^0x[a-fA-F0-9]{40}$",
                message = "Recipient must be a valid Ethereum address"
        )
        String recipient,

        @NotBlank(message = "Token address is required")
        @Pattern(
                regexp = "^0x[a-fA-F0-9]{40}$",
                message = "Token address must be valid"
        )
        String tokenAddress,

        @NotBlank(message = "Token symbol is required")
        @Size(max = 20)
        String tokenSymbol,

        @NotBlank(message = "Payment amount is required")
        @Pattern(
                regexp = "^[0-9]+(\\.[0-9]+)?$",
                message = "Amount must be a positive decimal number"
        )
        String amount

) {}
