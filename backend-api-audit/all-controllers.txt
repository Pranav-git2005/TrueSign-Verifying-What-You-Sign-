package com.truesign.controller;

import com.truesign.dto.request.CreatePassportRequest;
import com.truesign.dto.response.PassportResponse;
import com.truesign.service.PassportService;

import jakarta.validation.Valid;

import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.net.URI;

@RestController
@RequestMapping("/api/passports")
public class PassportController {

    private final PassportService passportService;

    public PassportController(PassportService passportService) {
        this.passportService = passportService;
    }

    @PostMapping
    public ResponseEntity<PassportResponse> createPassport(
            @Valid @RequestBody CreatePassportRequest request) {

        PassportResponse response =
                passportService.createPassport(request);

        return ResponseEntity
                .created(URI.create("/api/passports/" + response.passportId()))
                .body(response);
    }

    @GetMapping("/{passportId}")
    public ResponseEntity<PassportResponse> getPassport(
            @PathVariable String passportId) {

        return ResponseEntity.ok(
                passportService.getPassport(passportId)
        );
    }
}

package com.truesign.controller;

import com.truesign.dto.response.ProofBundleResponse;
import com.truesign.service.ExecutionReceiptService;
import com.truesign.service.ProofBundleService;
import org.springframework.http.HttpStatus;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.server.ResponseStatusException;

@RestController
@RequestMapping("/api/proofs")
public class ProofController {

    private final ProofBundleService proofBundleService;
    private final ExecutionReceiptService receiptService;

    public ProofController(
            ProofBundleService proofBundleService,
            ExecutionReceiptService receiptService
    ) {
        this.proofBundleService = proofBundleService;
        this.receiptService = receiptService;
    }

    public record ExecutionReceiptRequest(String transactionHash) {}

    @GetMapping("/{passportId}")
    public ProofBundleResponse getProof(
            @PathVariable String passportId
    ) {
        return proofBundleService.getLatestProofBundle(passportId);
    }

    @PostMapping("/{passportId}/execution")
    public ProofBundleResponse attachExecution(
            @PathVariable String passportId,
            @RequestBody ExecutionReceiptRequest request
    ) {
        if (request == null || request.transactionHash() == null) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Transaction hash is required."
            );
        }

        var receipt = receiptService.getExecutionReceipt(
                request.transactionHash()
        );

        if (receipt.isEmpty()) {
            throw new ResponseStatusException(
                    HttpStatus.ACCEPTED,
                    "Transaction is not mined yet. Retry after confirmation."
            );
        }

        return proofBundleService.attachExecutionReceipt(
                passportId,
                receipt.get()
        );
    }
}

        );

package com.truesign.controller;

import com.truesign.dto.request.VerifyTransactionRequest;
import com.truesign.dto.response.VerificationResponse;
import com.truesign.service.VerificationService;
import jakarta.validation.Valid;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/verify")
public class VerificationController {

    private final VerificationService verificationService;

    public VerificationController(
            VerificationService verificationService) {
        this.verificationService = verificationService;
    }

    // POST /api/verify
    @PostMapping
    public ResponseEntity<VerificationResponse> verify(
            @Valid @RequestBody VerifyTransactionRequest request) {

        VerificationResponse response =
                verificationService.verifyPassport(request.passportId());

        return ResponseEntity.ok(response);
    }

    // POST /api/verify/{passportId}
    @PostMapping("/{passportId}")
    public ResponseEntity<VerificationResponse> verifyByPassportId(
            @PathVariable String passportId) {

        VerificationResponse response =
                verificationService.verifyPassport(passportId);

        return ResponseEntity.ok(response);
    }
}
